      if (!binId) return;

      try {
        const headers = {};
        if (apiKey) headers['X-Master-Key'] = apiKey;

        const res = await fetch(`https://api.jsonbin.io/v3/b/${binId}/latest`, { headers });
        if (res.ok) {
          const json = await res.json();
          const data = json.record;
          if (data && Array.isArray(data.golfRounds)) {

          }
        }
      } catch (err) {
        console.warn('雲端擷取失敗，使用本機資料:', err);
      }
    }

    /**
     * 安全算式解析器：支援如 "600+600" 或 "400+100" 的計算
     */
    function parseAmountExpression(expr) {
      if (typeof expr === 'number') return isNaN(expr) ? 0 : expr;
      if (!expr || typeof expr !== 'string') return 0;

      const cleanExpr = expr.replace(/[$,\s]/g, '');
      if (!cleanExpr) return 0;

      if (!/^[\d+\-*/().]+$/.test(cleanExpr)) {
        const fallback = parseFloat(cleanExpr);
        return isNaN(fallback) ? 0 : Math.round(fallback);
      }

      try {
        const result = Function(`'use strict'; return (${cleanExpr})`)();
        return isNaN(result) ? 0 : Math.round(result);
      } catch (err) {
        const fallback = parseFloat(cleanExpr);
        return isNaN(fallback) ? 0 : Math.round(fallback);
      }
    }

    /**
     * 貨幣格式化函數
     */
    function formatCurrency(amount) {
      const val = Math.round(amount || 0);
      return (val < 0 ? '-$' : '$') + Math.abs(val).toLocaleString('en-US');
    }

    /**
     * 核心財務結算與金主排行榜計算
     */
    function calculateFinance() {
      let totalSponsorship = 0;
      let totalDeductions = 0;
      const playerMap = {};

      (window.golfRounds || []).forEach(round => {
        totalDeductions += (round.deduction || 0);

        (round.players || []).forEach(player => {
          const amt = player.amount || 0;
          totalSponsorship += amt;

          if (!playerMap[player.name]) {
            playerMap[player.name] = {
              name: player.name,
              totalAmount: 0,
              appearances: 0
            };
          }
          playerMap[player.name].totalAmount += amt;
          playerMap[player.name].appearances += 1;
        });
      });

      const totalBanquet = (window.banquetExpenses || []).reduce((sum, b) => sum + (b.amount || 0), 0);
      const netAvailableFund = window.initialBalance + totalSponsorship - totalDeductions - totalBanquet;

      const rankedPlayers = Object.values(playerMap).sort((a, b) => {
        if (b.totalAmount !== a.totalAmount) {
          return b.totalAmount - a.totalAmount;
        }
        return b.appearances - a.appearances;
      });

      return {
        initialBalance: window.initialBalance,
        totalSponsorship,
        totalDeductions,
        totalBanquet,
        netAvailableFund,
        rankedPlayers,
        totalRoundsCount: (window.golfRounds || []).length
      };
    }

    function handleSaveRound(e) {
      e.preventDefault();
      if (!isAdmin) return;

      const id = document.getElementById('round-id').value;
      const date = document.getElementById('round-date').value;
      const course = document.getElementById('round-course').value.trim();
      const deduction = parseAmountExpression(document.getElementById('round-deduction').value);
      const note = document.getElementById('round-note').value.trim();

      const nameInputs = document.querySelectorAll('.player-name-input');
      const exprInputs = document.querySelectorAll('.player-expr-input');

      const players = [];
      for (let i = 0; i < nameInputs.length; i++) {
        const name = nameInputs[i].value.trim();
        const expr = exprInputs[i].value.trim();
        if (name) {
          const amount = parseAmountExpression(expr);
          players.push({
            name,
            expr: expr || amount.toString(),
            amount
          });
        }
      }

      if (players.length === 0) {
        showToast('請至少輸入一位球友的贊助資料', '⚠️');
        return;
      }

      if (id) {
        const index = window.golfRounds.findIndex(r => r.id === id);
        if (index !== -1) {
          window.golfRounds[index] = { ...window.golfRounds[index], date, course, players, deduction, note };
          showToast('球敘紀錄已更新並儲存！');
        }
      } else {
        const newRound = {
          id: 'R-' + Date.now(),
          date,
          course,
          players,
          deduction,
          note
        };
        window.golfRounds.unshift(newRound);
        showToast('新球敘紀錄已成功建立！');
      }

      closeModal('modal-round');
      persistData();
      renderApp();
    }

        window.golfRounds = window.golfRounds.filter(r => r.id !== id);

        renderApp();
        showToast('球敘紀錄已刪除並更新存檔！');
      });
    }

    function editBanquet(id) {
      if (!isAdmin) {
        showToast('請先由右上角登入管理員以進行編輯', '🔒');
        return;
      }
      const b = window.banquetExpenses.find(x => x.id === id);

      document.getElementById('banquet-modal-title').innerHTML = `<span>✏️</span> 編輯餐敘開銷`;
      document.getElementById('banquet-title').value = b.title;
      document.getElementById('banquet-date').value = b.date;

      document.getElementById('banquet-note').value = b.note || '';

      openModal('modal-banquet');
    }

    function handleSaveBanquet(e) {
      e.preventDefault();
      if (!isAdmin) return;

      const id = document.getElementById('banquet-id').value;
      const title = document.getElementById('banquet-title').value.trim();
      const date = document.getElementById('banquet-date').value;
      const restaurant = document.getElementById('banquet-restaurant').value.trim();
      const amount = parseInt(document.getElementById('banquet-amount').value, 10) || 0;
      const note = document.getElementById('banquet-note').value.trim();

      if (id) {
        const idx = window.banquetExpenses.findIndex(x => x.id === id);
        if (idx !== -1) {
          window.banquetExpenses[idx] = { ...window.banquetExpenses[idx], title, date, restaurant, amount, note };
          showToast('餐敘紀錄已更新！');
        }
      } else {
        const newBanquet = {
          id: 'B-' + Date.now(),
          title,
          date,
          restaurant,
          amount,
          note
        };
        window.banquetExpenses.unshift(newBanquet);
        showToast('餐敘紀錄已建立！');
      }

      closeModal('modal-banquet');
      persistData();
      renderApp();
    }

    function deleteBanquet(id) {

    /**
     * 生成 LINE 群組專屬戰報
     */
    function generateLineReport() {
      const finance = calculateFinance();
      const latestRound = window.golfRounds.length > 0 ? [...window.golfRounds].sort((a,b) => new Date(b.date) - new Date(a.date))[0] : null;

      let report = `⛳【高球餐敘基金戰報】⛳\n`;
      report += `📅 統計結算：${new Date().toLocaleDateString('zh-TW')}\n`;
      report += `══════════════════\n`;
      report += `🌊 當前可用餐敘水庫：${formatCurrency(finance.netAvailableFund)}\n`;
      report += `💰 基金收入：${formatCurrency(finance.totalSponsorship)}\n`;
      report += `🧾 雜支：-${formatCurrency(finance.totalDeductions)}\n`;
      report += `🥩 餐敘開銷：-${formatCurrency(finance.totalBanquet)}\n`;
      report += `🏦 期初結餘：${formatCurrency(finance.initialBalance)}\n`;
      report += `🏌️‍♂️ 累積出賽：${finance.totalRoundsCount} 場\n`;
      report += `══════════════════\n`;
      report += `🏆【六大豪氣金主殿堂】\n`;

      const top6 = finance.rankedPlayers.slice(0, 6);
      const badges = ['🥇 榜首', '🥈 榜眼', '🥉 探花', '🎖️ 豪氣', '🌟 鐵桿', '🤝 鼎力'];
      top6.forEach((p, idx) => {
        report += `${badges[idx]} ${p.name}：${formatCurrency(p.totalAmount)} (${p.appearances}場)\n`;
      });

      if (latestRound) {
        report += `══════════════════\n`;
        report += `🏌️‍♂️【最近一場戰報：${latestRound.date} ${latestRound.course}】\n`;
        latestRound.players.forEach(p => {
          report += ` • ${p.name}：$${p.amount} ${p.expr !== p.amount.toString() ? '('+p.expr+')' : ''}\n`;
        });
        if (latestRound.deduction > 0) {
          report += ` • 雜支扣除：-$${latestRound.deduction}\n`;
        }
      }

      report += `══════════════════\n`;
      report += `🍻 打得越好・贊助越多・餐敘越澎湃！兄弟們繼續衝！🏌️‍♂️💨`;

      const textarea = document.getElementById('line-report-content');
      textarea.value = report;
      const status = document.getElementById('copy-status');
      if (status) status.classList.add('opacity-0');

      openModal('modal-line-report');
    }

    /**
     * 複製 LINE 戰報 (遵守 document.execCommand('copy') 規範)
     */
    function copyLineReport() {
      const textarea = document.getElementById('line-report-content');
      textarea.select();
      textarea.setSelectionRange(0, 99999);

      let success = false;
      try {
        success = document.execCommand('copy');
      } catch (err) {
        success = false;
      }

      const status = document.getElementById('copy-status');
      if (status) {
        if (success) {
          status.textContent = '✅ 複製成功，快貼到群組吧！';
          status.classList.remove('opacity-0');
          showToast('戰報已複製至剪貼簿！');
        } else {
          status.textContent = '⚠️ 請手動選取文字長按複製';
          status.classList.remove('opacity-0');
        }
      }
    }

    /**
     * 彈出視窗開關輔助
     */
    function openModal(id) {
      const m = document.getElementById(id);
      if (m) {
        m.classList.remove('hidden');
        if (id === 'modal-initial-balance') {
          document.getElementById('input-initial-balance').value = window.initialBalance;
        }
      }
    }

    function closeModal(id) {
      const m = document.getElementById(id);
      if (m) m.classList.add('hidden');
    }

    /**
     * Toast 提示盒
     */
    function showToast(msg, icon = '✅') {
      const toast = document.getElementById('toast');
      const toastMsg = document.getElementById('toast-msg');
      const toastIcon = document.getElementById('toast-icon');

      toastMsg.textContent = msg;
      toastIcon.textContent = icon;

      toast.classList.remove('translate-y-20', 'opacity-0');
      setTimeout(() => {
        toast.classList.add('translate-y-20', 'opacity-0');
      }, 2500);
    }
